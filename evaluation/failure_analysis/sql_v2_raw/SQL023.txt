QUESTION:
Show the five origin airports with the largest deterioration in on-time arrival performance from 2024 to 2025, requiring at least 5000 departures in each year.

GENERATED SQL:
WITH cte_2024 AS (
    SELECT 
        Origin, 
        ROUND(100.0 * SUM(CASE WHEN is_on_time_arrival THEN 1 ELSE 0 END) / COUNT(*), 2) AS on_time_2024,
        COUNT(*) AS departures_2024
    FROM fact_flights
    WHERE year = 2024
    GROUP BY Origin
    HAVING COUNT(*) >= 5000
),
cte_2025 AS (
    SELECT 
        Origin, 
        ROUND(100.0 * SUM(CASE WHEN is_on_time_arrival THEN 1 ELSE 0 END) / COUNT(*), 2) AS on_time_2025,
        COUNT(*) AS departures_2025
    FROM fact_flights
    WHERE year = 2025
    GROUP BY Origin
    HAVING COUNT(*) >= 5000
)
SELECT 
    cte_2025.Origin,
    ROUND(cte_2025.on_time_2025 - cte_2024.on_time_2024, 2) AS signed_change
FROM cte_2024
JOIN cte_2025 ON cte_2024.Origin = cte_2025.Origin
ORDER BY signed_change ASC
LIMIT 5;
